### 1.5 航点任务规划

```python
def upload_mission(waypoints):
    """
    上传航点任务
    
    waypoints: [(lat, lon, alt), ...]
    """
    print("上传航点任务...")
    
    # 清除现有任务
    master.mav.mission_clear_all_send(
        master.target_system,
        master.target_component
    )
    
    # 上传航点
    for i, (lat, lon, alt) in enumerate(waypoints):
        master.mav.mission_item_int_send(
            master.target_system,
            master.target_component,
            i,  # 序号
            mavutil.mavlink.MAV_FRAME_GLOBAL_RELATIVE_ALT,
            mavutil.mavlink.MAV_CMD_NAV_WAYPOINT,
            2, 0, 0, 0, 0, 0,
            int(lat * 1e7),
            int(lon * 1e7),
            alt
        )
        print(f"航点{i+1}: {lat}, {lon}, {alt}m")
    
    # 设置总航点数
    master.mav.mission_count_send(
        master.target_system,
        master.target_component,
        len(waypoints)
    )
    
    # 开始任务
    print("开始任务...")
    master.mav.command_long_send(
        master.target_system,
        master.target_component,
        mavutil.mavlink.MAV_CMD_MISSION_START,
        0, 0, 0, 0, 0, 0, 0, 0
    )

# 示例：上传 3 个航点
waypoints = [
    (30.1234, 120.5678, 50),
    (30.1245, 120.5689, 50),
    (30.1256, 120.5700, 50),
]
upload_mission(waypoints)
```

### 1.6 日志下载

```python
def download_logs():
    """下载飞控日志"""
    print("请求日志列表...")
    
    master.mav.log_request_list_send(
        master.target_system,
        master.target_component,
        0,  # 起始 ID
        10  # 数量
    )
    
    # 接收日志列表
    while True:
        msg = master.recv_match(type=['LOG_ENTRY'], blocking=True, timeout=5)
        if msg:
            print(f"日志{msg.id}: {msg.num_logs}条")
            break
    
    # 下载日志
    print("下载日志...")
    master.mav.log_request_data_send(
        master.target_system,
        master.target_component,
        0,  # 日志 ID
        0,  # 偏移
        0xffffffff  # 大小
    )
    
    # 保存日志
    with open('flight.log', 'wb') as f:
        while True:
            msg = master.recv_match(type=['LOG_DATA'], blocking=True, timeout=5)
            if msg:
                f.write(msg.data[:msg.count])
                if msg.count < len(msg.data):
                    break

download_logs()
```

### 2.2 连接飞控

```java
import com.MAVLink.MAVLinkPacket;
import com.MAVLink.Messages.MAVLinkMessage;
import com.MAVLink.Messages.ardupilotmega.*;

import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;

public class DroneController {
    private DatagramSocket socket;
    private InetAddress address;
    private int port = 14550;
    
    public DroneController() throws Exception {
        socket = new DatagramSocket(port);
        address = InetAddress.getByName("127.0.0.1");
    }
    
    public void connect() throws Exception {
        System.out.println("等待连接...");
        
        byte[] buffer = new byte[1024];
        DatagramPacket packet = new DatagramPacket(buffer, buffer.length);
        
        while (true) {
            socket.receive(packet);
            MAVLinkPacket mavPacket = MAVLinkPacket.decode(packet.getData());
            
            if (mavPacket != null) {
                MAVLinkMessage msg = mavPacket.unpack();
                if (msg.msgid == 0) {  // HEARTBEAT
                    msg_heartbeat heartbeat = (msg_heartbeat) msg;
                    System.out.println("连接成功！");
                    System.out.println("系统类型：" + heartbeat.type);
                    System.out.println("自动驾驶仪：" + heartbeat.autopilot);
                    break;
                }
            }
        }
    }
}
```

```java
public void listenTelemetry() throws Exception {
    byte[] buffer = new byte[1024];
    DatagramPacket packet = new DatagramPacket(buffer, buffer.length);
    
    while (true) {
        socket.receive(packet);
        MAVLinkPacket mavPacket = MAVLinkPacket.decode(packet.getData());
        
        if (mavPacket != null) {
            MAVLinkMessage msg = mavPacket.unpack();
            
            if (msg.msgid == 33) {  // GLOBAL_POSITION_INT
                msg_global_position_int pos = (msg_global_position_int) msg;
                double lat = pos.lat / 1e7;
                double lon = pos.lon / 1e7;
                double alt = pos.alt / 1000.0;
                System.out.printf("位置：%.6f, %.6f, 高度：%.1fm%n", lat, lon, alt);
            
            } else if (msg.msgid == 30) {  // ATTITUDE
                msg_attitude att = (msg_attitude) msg;
                System.out.printf("姿态：横滚=%.2f, 俯仰=%.2f, 偏航=%.2f%n",
                    att.roll, att.pitch, att.yaw);
            
            } else if (msg.msgid == 147) {  // BATTERY_STATUS
                msg_battery_status batt = (msg_battery_status) msg;
                double voltage = batt.voltages[0] / 1000.0;
                System.out.printf("电池电压：%.1fV%n", voltage);
            }
        }
    }
}
```

### 2.4 发送控制命令

```java
import com.MAVLink.Common.MAV_CMD;
import com.MAVLink.Messages.ardupilotmega.msg_command_long;

public void arm() {
    msg_command_long cmd = new msg_command_long();
    cmd.target_system = 1;
    cmd.target_component = 1;
    cmd.command = MAV_CMD.MAV_CMD_COMPONENT_ARM_DISARM;
    cmd.param1 = 1;  // 1=解锁，0=上锁
    
    MAVLinkPacket packet = cmd.pack();
    sendData(packet.encode());
}

public void takeoff(float altitude) {
    msg_command_long cmd = new msg_command_long();
    cmd.target_system = 1;
    cmd.target_component = 1;
    cmd.command = MAV_CMD.MAV_CMD_NAV_TAKEOFF;
    cmd.param7 = altitude;  // 目标高度
    
    MAVLinkPacket packet = cmd.pack();
    sendData(packet.encode());
}

private void sendData(byte[] data) {
    try {
        DatagramPacket packet = new DatagramPacket(data, data.length, address, port);
        socket.send(packet);
    } catch (Exception e) {
        e.printStackTrace();
    }
}
```

### 3.2 连接与通信

```javascript
const { mavlink } = require('mavlink');

// 创建连接
const connection = mavlink.createConnection({
    host: '127.0.0.1',
    port: 14550
});

connection.on('connect', () => {
    console.log('连接成功！');
});

// 监听心跳
connection.on('HEARTBEAT', (msg) => {
    console.log('收到心跳包');
    console.log('系统类型:', msg.type);
    console.log('自动驾驶仪:', msg.autopilot);
});

// 监听位置
connection.on('GLOBAL_POSITION_INT', (msg) => {
    const lat = msg.lat / 1e7;
    const lon = msg.lon / 1e7;
    const alt = msg.alt / 1000.0;
    console.log(`位置：${lat}, ${lon}, 高度：${alt}m`);
});

// 监听姿态
connection.on('ATTITUDE', (msg) => {
    console.log(`姿态：横滚=${msg.roll}, 俯仰=${msg.pitch}, 偏航=${msg.yaw}`);
});
```

### 4.2 接收消息

```cpp
#include <mavlink/common/mavlink.h>
#include <iostream>
#include <cstring>

int main() {
    uint8_t buffer[MAVLINK_MAX_PACKET_LEN];
    mavlink_message_t msg;
    mavlink_status_t status;
    
    int fd = open("/dev/ttyUSB0", O_RDWR);
    
    while (true) {
        int len = read(fd, buffer, sizeof(buffer));
        
        for (int i = 0; i < len; i++) {
            if (mavlink_parse_char(MAVLINK_COMM_0, buffer[i], &msg, &status)) {
                if (msg.msgid == MAVLINK_MSG_ID_GLOBAL_POSITION_INT) {
                    mavlink_global_position_int_t pos;
                    mavlink_msg_global_position_int_decode(&msg, &pos);
                    
                    std::cout << "位置：" 
                              << pos.lat / 1e7 << ", " 
                              << pos.lon / 1e7 << ", "
                              << pos.alt / 1000.0 << "m" 
                              << std::endl;
                }
            }
        }
    }
    
    return 0;
}
```

## 五、常见问题

### Q1：连接失败怎么办？

**答：**
- 检查端口号是否正确
- 检查波特率（通常 57600 或 115200）
- 检查防火墙设置
- 检查设备权限（`sudo chmod 666 /dev/ttyUSB0`）

### Q2：消息解析失败？

**答：**
- 确认 MAVLink 版本（1.0/2.0）
- 检查字节序（大小端）
- 确认消息 ID 映射

### Q3：控制无响应？

**答：**
- 确认无人机已解锁
- 确认模式正确
- 检查安全开关
- 检查遥控器优先级
